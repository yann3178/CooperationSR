{ ============================================================================
  Strategie "Little Rizzie" (Marci Silfrain) - MultiCharts EasyLanguage
  Long uniquement. Version simplifiee.

  Detection des swings : fonctions natives PivotHighVS / PivotLowVS. MultiCharts
  gere lui-meme le delai de confirmation des pivots (Strength barres a droite),
  donc AUCUNE gestion manuelle du lookahead n'est necessaire.

  Logique (decisions a la cloture de la barre) :
    - Bandes de Bollinger (BBLength, BBStdDevs) sur le Close.
    - Swing High / Low : pivots de force Strength = (SwingWindow-1)/2.
    - Structure "bear" des que Close < dernier swing low. En phase bear,
      BBreach=true des qu'un Close <= bande basse.
    - ENTREE long : structure bear ET BBreach ET Close > dernier swing high (BoS)
      ET ADX > seuil ET (option) Confluence Fibonacci ET (option) risque <= % capital.
    - SORTIE : Close < dernier swing low (trailing stop structurel).

  Fibonacci : sur MacroLookback barres, Macro_High/Macro_Low ; valide si le High
    survient apres le Low. Niveaux 38.2/50/61.8 %. Cible Rizzie = SwingLow -
    (SwingHigh - SwingLow). Confluence si la cible est a +/- FibTolerancePct %.

  Risque : (Close - dernier swing low) * BigPointValue * NumContracts
    <= RiskPctPerTrade % du capital disponible (InitialCapital + NetProfit).

  Note : ordres "this bar on close" pour coller au backtest (entree/sortie au
  close du signal). MaxBarsBack : Auto, ou >= MacroLookback + SwingWindow + 30.
============================================================================ }

Inputs:
    BBLength( 20 ),
    BBStdDevs( 2.0 ),
    SwingWindow( 5 ),          { impair ; Strength = (SwingWindow-1)/2 }
    ADXLength( 14 ),
    ADXThreshold( 20 ),
    UseFibFilter( true ),
    FibTolerancePct( 3.0 ),
    MacroLookback( 200 ),
    UseRiskFilter( true ),
    RiskPctPerTrade( 5.0 ),
    InitialCapital( 300000 ),
    NumContracts( 1 );

Variables:
    Strength( 0 ),
    BBmid( 0 ),
    BBlower( 0 ),
    ADXValue( 0 ),
    SHval( 0 ),
    SLval( 0 ),
    LastSwingHigh( 0 ),
    LastSwingLow( 0 ),
    HaveSH( false ),
    HaveSL( false ),
    MacroHigh( 0 ),
    MacroLow( 0 ),
    MacroRng( 0 ),
    MacroBull( false ),
    Fib382( 0 ),
    Fib50( 0 ),
    Fib618( 0 ),
    IsBear( false ),
    BBreach( false ),
    InPos( false ),
    RizzieTarget( 0 ),
    TradeRisk( 0 ),
    AvailCap( 0 ),
    CondFib( false ),
    CondRisk( false ),
    tol( 0 );

Once
    Strength = IntPortion( ( SwingWindow - 1 ) / 2 );

{ ---------- Indicateurs ---------- }
BBmid    = Average( Close, BBLength );
BBlower  = BBmid - BBStdDevs * StandardDev( Close, BBLength, 1 );
ADXValue = ADX( ADXLength );

{ ---------- Swings (pivots natifs MultiCharts, confirmes par MC) ---------- }
SHval = PivotHighVS( 1, High, Strength, Strength, 1 );
SLval = PivotLowVS(  2, Low,  Strength, Strength, 1 );
if SHval <> -1 then begin LastSwingHigh = SHval; HaveSH = true; end;
if SLval <> -1 then begin LastSwingLow  = SLval; HaveSL = true; end;

{ ---------- Zones de retracement Fibonacci (macro bull run) ---------- }
MacroHigh = Highest( High, MacroLookback );
MacroLow  = Lowest( Low, MacroLookback );
MacroRng  = MacroHigh - MacroLow;
MacroBull = ( MacroRng > 0 ) and ( HighestBar( High, MacroLookback ) < LowestBar( Low, MacroLookback ) );
if MacroRng > 0 then
begin
    Fib382 = MacroHigh - 0.382 * MacroRng;
    Fib50  = MacroHigh - 0.500 * MacroRng;
    Fib618 = MacroHigh - 0.618 * MacroRng;
end;

{ ---------- 1) SORTIE ---------- }
if InPos then
begin
    if HaveSL and ( Close < LastSwingLow ) then
    begin
        Sell ( "RizzieLX" ) NumContracts contracts this bar on close;
        InPos   = false;
        IsBear  = true;
        BBreach = false;
    end;
end;

{ ---------- 2) Structure + suivi bande basse (a plat) ---------- }
if InPos = false then
begin
    if HaveSL and ( Close < LastSwingLow ) and ( IsBear = false ) then
    begin
        IsBear  = true;
        BBreach = false;
    end;
    if IsBear and ( Close <= BBlower ) then
        BBreach = true;
end;

{ ---------- 3) ENTREE (a plat) ---------- }
if InPos = false then
begin
    { Filtre Confluence Fibonacci (optionnel) }
    CondFib = true;
    if UseFibFilter then
    begin
        CondFib = false;
        if MacroBull and HaveSH and HaveSL then
        begin
            RizzieTarget = LastSwingLow - ( LastSwingHigh - LastSwingLow );
            if RizzieTarget > 0 then
            begin
                tol = FibTolerancePct / 100;
                if ( Fib382 > 0 ) and ( AbsValue( RizzieTarget - Fib382 ) / Fib382 <= tol ) then CondFib = true;
                if ( Fib50  > 0 ) and ( AbsValue( RizzieTarget - Fib50  ) / Fib50  <= tol ) then CondFib = true;
                if ( Fib618 > 0 ) and ( AbsValue( RizzieTarget - Fib618 ) / Fib618 <= tol ) then CondFib = true;
            end;
        end;
    end;

    { Filtre de risque par trade (optionnel) }
    CondRisk = true;
    if UseRiskFilter then
    begin
        CondRisk = false;
        if HaveSL and ( Close > LastSwingLow ) then
        begin
            TradeRisk = ( Close - LastSwingLow ) * BigPointValue * NumContracts;
            AvailCap  = InitialCapital + NetProfit;
            if TradeRisk <= ( RiskPctPerTrade / 100 ) * AvailCap then CondRisk = true;
        end;
    end;

    if IsBear and BBreach
        and HaveSH and ( Close > LastSwingHigh )
        and ( ADXValue > ADXThreshold )
        and CondFib and CondRisk then
    begin
        Buy ( "RizzieLE" ) NumContracts contracts this bar on close;
        InPos   = true;
        IsBear  = false;
        BBreach = false;
    end;
end;
