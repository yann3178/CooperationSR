{*******************************************************************************
 * Function: YANN_RENKO_CheckShortEntry
 * Version: 2.0.1
 * Description: Checks if short entry conditions are met for Renko strategy
 *
 * Returns: True if short entry signal is valid, False otherwise
 *******************************************************************************}

Inputs:
	CurrentBrickColor(NumericSimple),      // 1=GREEN, -1=RED, 0=NEUTRAL
	PreviousBrickColor(NumericSimple),     // Previous brick color
	EntryConfirmType(StringSimple),        // "FIRST_GREEN" or "SECOND_GREEN"
	RequirePriorRed(TrueFalse),            // Require previous brick to be opposite (GREEN)
	AllowShort(TrueFalse),                 // Short trades allowed
	UseTrendFilter(TrueFalse),             // Use trend filter
	CurrentPrice(NumericSimple),           // Current close price
	TrendSMA(NumericSimple),               // Trend SMA value
	TrendMargin(NumericSimple);            // Trend margin (e.g., 0.01 for 1%)

Variables:
	var0(false),                           // Signal flag
	TrendOK(false),
	EntryConditionMet(false);

// Initialize
var0 = false;

// Check if short trades allowed
If AllowShort = false Then Begin
	YANN_RENKO_CheckShortEntry = false;
	Return;
End;

// ========================================
// Check Trend Filter (if enabled)
// ========================================
TrendOK = true;

If UseTrendFilter Then Begin
	// For SHORT: Price must be below SMA - margin
	TrendOK = CurrentPrice < (TrendSMA * (1 - TrendMargin));
End;

If TrendOK = false Then Begin
	YANN_RENKO_CheckShortEntry = false;
	Return;
End;

// ========================================
// Check Entry Confirmation
// ========================================
EntryConditionMet = false;

// FIRST_RED: Current brick is RED (after optional GREEN check)
If EntryConfirmType = "FIRST_GREEN" Then Begin
	// Note: For SHORT, we use "FIRST_RED" as equivalent to "FIRST_GREEN" for longs
	If CurrentBrickColor = -1 Then Begin
		If RequirePriorRed Then Begin
			// For shorts, require previous brick to be GREEN (opposite)
			EntryConditionMet = (PreviousBrickColor = 1);
		End
		Else Begin
			// Any previous color OK
			EntryConditionMet = true;
		End;
	End;
End

// SECOND_RED: Current and previous both RED
Else If EntryConfirmType = "SECOND_GREEN" Then Begin
	// Note: For SHORT, we use "SECOND_RED" as equivalent to "SECOND_GREEN" for longs
	If CurrentBrickColor = -1 And PreviousBrickColor = -1 Then Begin
		EntryConditionMet = true;
	End;
End;

// Return result
YANN_RENKO_CheckShortEntry = EntryConditionMet;
